# B276 - Create campy uninstall Command and Uninstall Script

State: complete -- verified via git log 2026-08-02 (feat: implement B265-B276 215b588); card file was stale
Priority: P1

Plan: backlog/plans/B-276-uninstaller.md

### Problem

Campy installs artifacts across 10+ locations on the user's machine. There's no way to cleanly uninstall — users must manually hunt down launchd plists, config file entries, skill directories, hook files, MCP registrations, and the Python package. We need this both for users AND for testing the new installer (B272).

### What This Card Does

1. Creates a `campy uninstall` CLI command that removes all Campy artifacts
2. Creates a standalone `scripts/uninstall.sh` for users who can't run `campy` (e.g., broken install)

### Dependencies

- None (can be implemented immediately)

### Complete Inventory of Campy Artifacts

Based on audit of a real install, Campy touches ALL of these locations:

**1. Daemon (must stop first)**
- Process: `brain_daemon.py` (kill via `pkill -f brain_daemon`)
- Launchd plists:
  - `~/Library/LaunchAgents/ai.hippocampy.brain.plist`
  - `~/Library/LaunchAgents/ai.sidequests.mission-control.plist` (legacy)
  - `~/Library/LaunchAgents/dev.hippocampy.indicator.plist`

**2. Campy data directory**
- `~/.campy/` — contains daemon.log, activity.log, triggers/, KuzuDB graph data, offline queue
- This is where ALL user memory lives. Uninstall should WARN before deleting this.

**3. Claude Code**
- MCP registration: entry in `~/.claude/settings.json` under `mcpServers`
- Hook entries: `hooks.UserPromptSubmit` entries containing `hook_user_turn.py`
- Project hooks: `.claude/hooks/` in any project where `campy setup` was run
- Plugin (if native install): `~/.claude/plugins/hippocampy/` or plugin cache

**4. Claude Desktop**
- MCP entry in `~/Library/Application Support/Claude/claude_desktop_config.json`

**5. Codex**
- Skills: `~/.codex/skills/` — Campy skill directories (recall, brief, learn, campy-grill, etc.)
- Hooks: `~/.codex/hooks/campy/` directory
- Hook config: entries in `~/.codex/hooks.json` containing "campy"
- MCP config: `[mcp_servers.campy]` block in `~/.codex/config.toml`
- Archived skills: `~/.codex/skills/_archived_by_campy/`
- Plugin (if native install): `~/.codex/plugins/hippocampy/` or plugin cache

**6. Gemini CLI**
- Skills: `~/.gemini/skills/` — Campy skill directories
- Hooks: `~/.gemini/hooks/campy/` directory
- Hook config: entries in `~/.gemini/settings.json` hooks section containing "campy"
- MCP config: entries in `~/.gemini/settings.json` mcpServers section
- Context file: `GEMINI.md` in installed package or project
- Extension (if native install): linked or cached extension

**7. VS Code**
- MCP entry in `~/Library/Application Support/Code/User/mcp.json`

**8. OpenClaw**
- Config entries (check `~/.openclaw/` or equivalent)

**9. Python package**
- pipx: `pipx uninstall hippocampy`
- pip: `pip uninstall hippocampy`
- Console scripts: `campy`, `campy-daemon`, `sidequests`, `sidequests-daemon`

"""Campy uninstaller — removes all Campy artifacts from the system."""
import json
import os
import re
import shutil
import subprocess
import sys
from pathlib import Path

from rich.console import Console
from rich.prompt import Confirm

console = Console()

# All known Campy skill directory names
CAMPY_SKILL_NAMES = {
    "recall", "brief", "learn", "session-start", "memory-awareness",
    "quest-management", "status", "campy-memory",
    "campy-grill", "campy-diagnose", "campy-tdd",
    "campy-handoff", "campy-improve-architecture",
    # Legacy names (before rename)
    "grill", "diagnose", "tdd", "handoff", "improve-architecture",
}

LAUNCHD_LABELS = [
    "ai.hippocampy.brain",
    "ai.sidequests.mission-control",
    "dev.hippocampy.indicator",
]


def uninstall(keep_data: bool = False, force: bool = False) -> None:
    """Remove all Campy artifacts from the system.
    
    Args:
        keep_data: If True, preserve ~/.campy/ (graph data, memories).
        force: If True, skip confirmation prompts.
    """
    console.print("[bold red]Campy Uninstaller[/bold red]")
    console.print("")
    
    # Inventory what we'll remove
    artifacts = _inventory()
    if not artifacts:
        console.print("[green]No Campy artifacts found. Nothing to uninstall.[/green]")
        return
    
    console.print("[yellow]Found these Campy artifacts:[/yellow]")
    for category, items in artifacts.items():
        console.print(f"  [bold]{category}[/bold]")
        for item in items:
            console.print(f"    • {item}")
    console.print("")
    
    if not keep_data and any("~/.campy" in str(i) for items in artifacts.values() for i in items):
        console.print("[bold red]⚠ WARNING: This will DELETE your Campy memory graph![/bold red]")
        console.print("[dim]  Use --keep-data to preserve ~/.campy/[/dim]")
        console.print("")
    
    if not force:
        if not Confirm.ask("Proceed with uninstall?"):
            console.print("[dim]Cancelled.[/dim]")
            return
    
    # Step 1: Stop daemon
    _stop_daemon()
    
    # Step 2: Remove launchd plists
    _remove_launchd()
    
    # Step 3: Remove Claude Code artifacts
    _remove_claude_code()
    
    # Step 4: Remove Claude Desktop artifacts
    _remove_claude_desktop()
    
    # Step 5: Remove Codex artifacts
    _remove_codex()
    
    # Step 6: Remove Gemini artifacts
    _remove_gemini()
    
    # Step 7: Remove VS Code artifacts
    _remove_vscode()
    
    # Step 8: Remove data directory
    if not keep_data:
        _remove_data_dir()
    else:
        console.print("[dim]  Keeping ~/.campy/ (--keep-data)[/dim]")
    
    # Step 9: Remove Python package (must be last — this removes `campy` command)
    _remove_package()
    
    console.print("")
    console.print("[green]✓ Campy uninstalled.[/green]")
    if keep_data:
        console.print("[dim]  Memory data preserved at ~/.campy/[/dim]")

def _stop_daemon():
    """Stop the brain daemon."""
    console.print("[yellow]Stopping daemon...[/yellow]")
    for label in LAUNCHD_LABELS:
        subprocess.run(["launchctl", "unload",
            str(Path.home() / "Library" / "LaunchAgents" / f"{label}.plist")],
            capture_output=True)
    subprocess.run(["pkill", "-f", "brain_daemon"], capture_output=True)


def _remove_launchd():
    """Remove launchd plist files."""
    for label in LAUNCHD_LABELS:
        plist = Path.home() / "Library" / "LaunchAgents" / f"{label}.plist"
        if plist.exists():
            plist.unlink()
            console.print(f"  [red]✗[/red] Removed {plist}")


def _remove_claude_code():
    """Remove Campy from Claude Code settings."""
    settings = Path.home() / ".claude" / "settings.json"
    if settings.exists():
        try:
            data = json.loads(settings.read_text())
            changed = False
            # Remove MCP server
            if "campy" in data.get("mcpServers", {}):
                del data["mcpServers"]["campy"]
                changed = True
            # Remove hooks containing hook_user_turn.py
            hooks = data.get("hooks", {}).get("UserPromptSubmit", [])
            original_len = len(hooks)
            hooks[:] = [e for e in hooks
                        if not any("hook_user_turn" in str(h.get("command",""))
                                   for h in e.get("hooks",[]))]
            if len(hooks) != original_len:
                changed = True
            if changed:
                settings.write_text(json.dumps(data, indent=2))
                console.print(f"  [red]✗[/red] Cleaned {settings}")
        except Exception:
            pass
    # Remove project hooks
    project_hooks = Path.cwd() / ".claude" / "hooks"
    if project_hooks.exists():
        for f in project_hooks.glob("*.sh"):
            f.unlink()
        console.print(f"  [red]✗[/red] Removed project hooks from {project_hooks}")


def _remove_claude_desktop():
    """Remove Campy from Claude Desktop config."""
    config = Path.home() / "Library" / "Application Support" / "Claude" / "claude_desktop_config.json"
    if config.exists():
        try:
            data = json.loads(config.read_text())
            if "campy" in data.get("mcpServers", {}):
                del data["mcpServers"]["campy"]
                config.write_text(json.dumps(data, indent=2))
                console.print(f"  [red]✗[/red] Cleaned {config}")
        except Exception:
            pass


def _remove_codex():
    """Remove Campy skills, hooks, and MCP config from Codex."""
    # Skills
    skills_dir = Path.home() / ".codex" / "skills"
    if skills_dir.exists():
        for name in CAMPY_SKILL_NAMES:
            skill_path = skills_dir / name
            if skill_path.exists():
                shutil.rmtree(skill_path)
        # Remove archived skills
        archive = skills_dir / "_archived_by_campy"
        if archive.exists():
            shutil.rmtree(archive)
        console.print(f"  [red]✗[/red] Removed Codex skills")
    # Hooks
    hooks_dir = Path.home() / ".codex" / "hooks" / "campy"
    if hooks_dir.exists():
        shutil.rmtree(hooks_dir)
        console.print(f"  [red]✗[/red] Removed Codex hooks")
    # hooks.json — remove campy entries
    hooks_json = Path.home() / ".codex" / "hooks.json"
    if hooks_json.exists():
        try:
            data = json.loads(hooks_json.read_text())
            for event in list(data.get("hooks", {}).keys()):
                entries = data["hooks"][event]
                data["hooks"][event] = [
                    e for e in entries
                    if not any("campy" in str(h.get("command",""))
                               for h in e.get("hooks",[]))
                ]
            hooks_json.write_text(json.dumps(data, indent=2))
            console.print(f"  [red]✗[/red] Cleaned {hooks_json}")
        except Exception:
            pass
    # config.toml — remove [mcp_servers.campy] block
    config = Path.home() / ".codex" / "config.toml"
    if config.exists():
        try:
            content = config.read_text()
            # Remove [mcp_servers.campy] block
            pattern = re.compile(
                r'^\s*\[mcp_servers\.campy\]\s*\n(?:^[^\[][^\n]*\n?)*',
                re.MULTILINE,
            )
            cleaned = pattern.sub("", content).rstrip() + "\n"
            if cleaned != content:
                config.write_text(cleaned)
                console.print(f"  [red]✗[/red] Cleaned {config}")
        except Exception:
            pass


def _remove_gemini():
    """Remove Campy skills, hooks, and config from Gemini CLI."""
    # Skills
    skills_dir = Path.home() / ".gemini" / "skills"
    if skills_dir.exists():
        for name in CAMPY_SKILL_NAMES:
            skill_path = skills_dir / name
            if skill_path.exists():
                shutil.rmtree(skill_path)
        console.print(f"  [red]✗[/red] Removed Gemini skills")
    # Hooks
    hooks_dir = Path.home() / ".gemini" / "hooks" / "campy"
    if hooks_dir.exists():
        shutil.rmtree(hooks_dir)
        console.print(f"  [red]✗[/red] Removed Gemini hooks")
    # settings.json — remove campy entries from hooks and mcpServers
    settings = Path.home() / ".gemini" / "settings.json"
    if settings.exists():
        try:
            data = json.loads(settings.read_text())
            changed = False
            # Remove MCP server
            if "hippocampy" in data.get("mcpServers", {}):
                del data["mcpServers"]["hippocampy"]
                changed = True
            # Remove hook entries with campy in name
            for event in list(data.get("hooks", {}).keys()):
                entries = data["hooks"][event]
                original = len(entries)
                data["hooks"][event] = [
                    e for e in entries
                    if not any(h.get("name","").startswith("campy-")
                               for h in e.get("hooks",[]))
                ]
                if len(data["hooks"][event]) != original:
                    changed = True
            if changed:
                settings.write_text(json.dumps(data, indent=2))
                console.print(f"  [red]✗[/red] Cleaned {settings}")
        except Exception:
            pass


def _remove_vscode():
    """Remove Campy from VS Code MCP config."""
    mcp_json = Path.home() / "Library" / "Application Support" / "Code" / "User" / "mcp.json"

    data_dir = Path.home() / ".campy"

        shutil.rmtree(data_dir)
        console.print(f"  [red]✗[/red] Removed {data_dir}")


def _remove_package():
    """Remove the Python package."""
    console.print("[yellow]Removing Python package...[/yellow]")
    # Try pipx first
    result = subprocess.run(["pipx", "uninstall", "hippocampy"],
                          capture_output=True, text=True)
    if result.returncode == 0:
        console.print(f"  [red]✗[/red] Uninstalled via pipx")
    else:
        # Try pip
        subprocess.run([sys.executable, "-m", "pip", "uninstall", "-y", "hippocampy"],
                      capture_output=True)
        console.print(f"  [red]✗[/red] Uninstalled via pip")

# Standalone Campy uninstaller — use when `campy uninstall` isn't available

launchctl unload ~/Library/LaunchAgents/ai.sidequests.mission-control.plist 2>/dev/null || true
launchctl unload ~/Library/LaunchAgents/dev.hippocampy.indicator.plist 2>/dev/null || true

# Remove launchd plists
rm -f ~/Library/LaunchAgents/ai.hippocampy.brain.plist

rm -f ~/Library/LaunchAgents/dev.hippocampy.indicator.plist

rm -rf ~/.codex/skills/session-start ~/.codex/skills/memory-awareness

# Remove Gemini artifacts
rm -rf ~/.gemini/skills/recall ~/.gemini/skills/brief ~/.gemini/skills/learn
rm -rf ~/.gemini/skills/session-start ~/.gemini/skills/memory-awareness
rm -rf ~/.gemini/skills/quest-management ~/.gemini/skills/status

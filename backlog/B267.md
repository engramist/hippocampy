### Problem

Hook logic is duplicated across 7+ files in `adapters/claude_code/hooks/`, `adapters/codex/hooks/`, and `adapters/gemini_cli/hooks/`. Each platform has its own implementation of the same trigger manifest matching, but with different I/O protocols. This duplication caused the `_with_phase` bug to go unnoticed and makes maintenance error-prone.

### What This Card Does

Creates a single `plugin/hooks/campy_hook.py` module containing all shared hook logic (trigger manifest loading, pattern matching, context snippet formatting). Each platform gets a thin wrapper that handles its specific I/O protocol:

- **Claude Code**: Bash scripts that translate `$CLAUDE_TOOL_NAME`/`$CLAUDE_TOOL_INPUT` env vars → call `campy_hook.py` functions
- **Codex**: Thin Python scripts that read JSON stdin → call `campy_hook.py` → write JSON stdout  
- **Gemini**: Thin Python scripts that read JSON stdin → call `campy_hook.py` → write JSON stdout (different output schema)

### Files to Create

1. **`plugin/hooks/campy_hook.py`** (~120 lines) — The shared module. Contains:

```python
"""Shared hook logic for all platforms.

All platform-specific wrappers call these functions.
The module has ZERO platform-specific imports — it reads
trigger manifests and returns plain Python dicts.
"""
import json
import re
import os
import subprocess
import sys
from pathlib import Path

MANIFEST_PATH = Path.home() / ".campy" / "triggers" / "manifest.json"
MAX_MATCHES = 3

def load_manifest() -> list[dict]:
    """Load trigger manifest from ~/.campy/triggers/manifest.json."""
    if not MANIFEST_PATH.exists():
        return []
    try:
        return json.loads(MANIFEST_PATH.read_text())
    except (json.JSONDecodeError, OSError):
        return []

def match_triggers(
    manifest: list[dict],
    hook_type: str,
    tool_name: str = "",
    content: str = "",
) -> list[dict]:
    """Find matching triggers from the manifest.
    
    Args:
        manifest: Loaded trigger manifest entries.
        hook_type: "PreToolUse" or "PostToolUse".
        tool_name: The tool being called (e.g., "Bash", "Edit").
        content: The tool input or output to match against.
    
    Returns:
        List of matched trigger dicts (capped at MAX_MATCHES).
    """
    matches = []
    for trigger in manifest:
        if trigger.get("hook_type") != hook_type:
            continue
        # Tool name filter (if trigger specifies one)
        trigger_tool = trigger.get("tool_name", "")
        if trigger_tool and trigger_tool != tool_name:
            continue
        # Pattern match against content
        pattern = trigger.get("pattern", "")
        if pattern and not re.search(pattern, content, re.IGNORECASE):
            continue
        matches.append(trigger)
        if len(matches) >= MAX_MATCHES:
            break
    return matches

def format_context_snippets(matches: list[dict]) -> str:
    """Format matched triggers as context snippets for injection.
    
    Returns a string like:
        [Campy Memory — Lesson: deployment-safety]
        Always run smoke tests before deploying to prod.
        
        [Campy Memory — Procedure: retry-strategy]
        Use exponential backoff with jitter.
    """
    if not matches:
        return ""
    parts = []
    for m in matches:
        name = m.get("name", "unknown")
        node_type = m.get("node_type", "Memory")
        snippet = m.get("snippet", "")
        parts.append(f"[Campy Memory — {node_type}: {name}]\n{snippet}")
    return "\n\n".join(parts)

def get_session_context() -> str:
    """Get session start context from the Campy daemon.
    
    Calls `campy decide "new session starting" --format=prompt`.
    Returns empty string if daemon is not running.
    """
    try:
        result = subprocess.run(
            ["campy", "decide", "new session starting", "--format=prompt"],
            capture_output=True, text=True, timeout=10,
        )
        return result.stdout.strip() if result.returncode == 0 else ""
    except (FileNotFoundError, subprocess.TimeoutExpired):
        return ""

def check_daemon_health() -> bool:
    """Check if the Campy daemon is running."""
    try:
        import urllib.request
        req = urllib.request.Request("http://127.0.0.1:7799/health")
        with urllib.request.urlopen(req, timeout=2) as resp:
            return resp.getcode() == 200
    except Exception:
        return False
```

3. **`plugin/hooks/session_start_wrapper.py`** (~20 lines):
```python
#!/usr/bin/env python3
"""SessionStart hook wrapper — works for Claude Code and Codex."""
import sys, os
sys.path.insert(0, os.path.dirname(__file__))
from campy_hook import check_daemon_health, get_session_context

if not check_daemon_health():
    sys.exit(0)  # Silent exit if daemon not running

context = get_session_context()
if context:
    print(context)  # Plain text stdout for both Claude Code and Codex SessionStart
```

4. **`plugin/hooks/pre_tool_use_wrapper.py`** (~25 lines):
```python
#!/usr/bin/env python3
"""PreToolUse hook wrapper — reads JSON stdin, outputs context."""
import json, sys, os
sys.path.insert(0, os.path.dirname(__file__))
from campy_hook import load_manifest, match_triggers, format_context_snippets

# Read hook input from stdin (JSON for both Claude Code and Codex)
try:
    data = json.load(sys.stdin)
except (json.JSONDecodeError, EOFError):
    sys.exit(0)

tool_name = data.get("tool_name", os.environ.get("CLAUDE_TOOL_NAME", ""))
tool_input = json.dumps(data.get("tool_input", {}))

manifest = load_manifest()
if not manifest:
    sys.exit(0)

matches = match_triggers(manifest, "PreToolUse", tool_name, tool_input)
if not matches:
    sys.exit(0)

context = format_context_snippets(matches)
print(context)  # stdout for context injection
```

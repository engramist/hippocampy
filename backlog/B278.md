### Problem

The ARC Agent v2 architecture (see `ARC_AGI/docs/specs/refactorPlan.md`) needs 14 brain-region-aligned MCP query tools to query the existing ARC graph schema. B277 built the basal ganglia foundation modules (`action_selector.py`, `reward_predictor.py`, `exploration_policy.py`). This card adds the ARC-specific tools on top of B277 and the existing hippocampus/temporal lobe modules, then registers them in `TOOL_HANDLERS`.

All tools are bounded Cypher traversals over existing ARC node types (`GridEntity`, `ActionFact`, `ActionEffect`, `Hypothesis`, `VictoryCondition`, `ArcMechanic`, etc.) — no new node types or schema changes needed.

async def arc_perceive_state(params: dict, db, config: dict) -> dict:
    """Ingest a game observation. Create GridSnapshot, update GridEntity nodes, create ActionEffect.
    
    params: {task_id, step, grid_hash, entities: [{color_id, centroid_row, centroid_col, pixel_count, role}],
             action_taken, effect: {n_cells_changed, apparent_effect}}
    """
    task_id = params["task_id"]
    step = params["step"]
    
    # Create GridSnapshot
    snapshot_id = f"{task_id}_step{step}"
    await db.execute_write(
        "MERGE (s:GridSnapshot {snapshot_id: $sid}) "
        "SET s.task_id = $tid, s.step = $step, s.grid_hash = $hash, "
        "    s.n_entities = $n, s.created_at = timestamp()",
        {"sid": snapshot_id, "tid": task_id, "step": step,
         "hash": params.get("grid_hash", ""), "n": len(params.get("entities", []))}
    )
    
    # Upsert GridEntity nodes
    for ent in params.get("entities", []):
        entity_id = f"{task_id}_e{ent.get('color_id', 0)}_{ent.get('region_index', 0)}"
        await db.execute_write(
            "MERGE (e:GridEntity {entity_id: $eid}) "
            "SET e.task_id = $tid, e.color_id = $cid, "
            "    e.centroid_row = $cr, e.centroid_col = $cc, "
            "    e.pixel_count = $pc, e.inferred_role = $role, "
            "    e.last_updated_step = $step",
            {"eid": entity_id, "tid": task_id, "cid": ent.get("color_id"),
             "cr": ent.get("centroid_row"), "cc": ent.get("centroid_col"),
             "pc": ent.get("pixel_count"), "role": ent.get("role", "unknown"),
             "step": step}
        )
    
    # Create ActionEffect if action was taken
    if params.get("action_taken"):
        effect_id = f"{task_id}_step{step}_effect"
        await db.execute_write(
            "MERGE (ae:ActionEffect {effect_id: $eid}) "
            "SET ae.task_id = $tid, ae.action_id = $aid, ae.step = $step, "
            "    ae.n_cells_changed = $ncc, ae.apparent_effect = $eff",
            {"eid": effect_id, "tid": task_id, "aid": params["action_taken"],
             "step": step, "ncc": params.get("effect", {}).get("n_cells_changed", 0),
             "eff": params.get("effect", {}).get("apparent_effect", "unknown")}
        )
    
    return {"snapshot_id": snapshot_id, "entity_count": len(params.get("entities", [])),
            "delta_from_previous": None}  # delta computed by comparing snapshots
```

async def arc_get_game_context(params: dict, db, config: dict) -> dict:
    """Compact episodic summary of current game belief state.
    params: {task_id}
    """
    task_id = params["task_id"]
    
    # Count hypotheses
    hyp_result = db.execute(
        "MATCH (h:Hypothesis {task_id: $tid}) WHERE h.status = 'active' RETURN count(h)",
        {"tid": task_id})
    hyp_count = hyp_result.get_next()[0] if hyp_result.has_next() else 0
    
    # Get latest step
    step_result = db.execute(
        "MATCH (s:GridSnapshot {task_id: $tid}) RETURN max(s.step)", {"tid": task_id})
    latest_step = step_result.get_next()[0] if step_result.has_next() else 0
    
    # Get action summary
    action_result = db.execute(
        "MATCH (af:ActionFact {task_id: $tid}) RETURN af.action_id, af.value_status, af.confidence",
        {"tid": task_id})
    actions = {}
    while action_result.has_next():
        row = action_result.get_next()
        actions[row[0]] = {"value_status": row[1], "confidence": row[2]}
    
    return {"hypothesis_count": hyp_count, "step": latest_step,
            "action_summary": actions, "goal": None, "progress_trend": "unknown"}


async def arc_get_action_evidence(params: dict, db, config: dict) -> dict:
    """Action track record from ActionFact/ActionEffect history.
    params: {task_id, action_id}
    """
    task_id = params["task_id"]
    action_id = params["action_id"]
    
    result = db.execute(
        "MATCH (af:ActionFact {task_id: $tid, action_id: $aid}) "
        "RETURN af.fact_type, af.confidence, af.value_status, af.evidence_count, af.observation_count",
        {"tid": task_id, "aid": action_id})
    
    if result.has_next():
        row = result.get_next()
        return {"tested": True, "fact_type": row[0], "confidence": row[1] or 0,
                "value_status": row[2] or "unknown", "evidence_count": row[3] or 0,
                "steps_used": row[4] or 0, "falsified_count": 0, "causal_power": row[1] or 0}
    
    return {"tested": False, "falsified_count": 0, "causal_power": 0, "value_status": "untested", "steps_used": 0}


async def arc_get_untested_actions(params: dict, db, config: dict) -> dict:
    """List actions never tried in this game.
    params: {task_id, available_actions: [str]}
    """
    task_id = params["task_id"]
    available = params.get("available_actions", [])
    
    result = db.execute(
        "MATCH (af:ActionFact {task_id: $tid}) RETURN DISTINCT af.action_id", {"tid": task_id})
    tested = set()
    while result.has_next():
        tested.add(result.get_next()[0])
    
    untested = [a for a in available if a not in tested]
    return {"untested": untested, "tested": list(tested)}


async def arc_get_causal_path(params: dict, db, config: dict) -> dict:
    """Bounded traversal: does this action lead toward the goal?
    params: {task_id, action_id, goal_id}
    """
    task_id = params["task_id"]
    action_id = params["action_id"]
    goal_id = params.get("goal_id", "")
    
    # Bounded 4-hop traversal: ActionFact -> ActionEffect -> MOVED_BY <- GridEntity -> REQUIRES_ENTITY <- VictoryCondition
    result = db.execute(
        "MATCH (af:ActionFact {task_id: $tid, action_id: $aid})-[:DERIVED_FROM_FACT]->(ae:ActionEffect)"
        "<-[:MOVED_BY]-(ge:GridEntity)-[:REQUIRES_ENTITY]-(vc:VictoryCondition {task_id: $tid}) "
        "RETURN count(*) as path_count, min(vc.confidence) as min_conf",
        {"tid": task_id, "aid": action_id})
    
    if result.has_next():
        row = result.get_next()
        path_count = row[0] or 0
        return {"path_exists": path_count > 0, "path_length": 4 if path_count > 0 else 0,
                "path_confidence": row[1] or 0}
    
    return {"path_exists": False, "path_length": 0, "path_confidence": 0}


async def arc_record_action_effect(params: dict, db, config: dict) -> dict:
    """Write: what happened when we took this action.
    params: {task_id, action_id, step, effect: {n_cells_changed, apparent_effect}, entities_affected: []}
    """
    task_id = params["task_id"]
    action_id = params["action_id"]
    step = params["step"]
    effect = params.get("effect", {})
    
    effect_id = f"{task_id}_{action_id}_step{step}"
    await db.execute_write(
        "MERGE (ae:ActionEffect {effect_id: $eid}) "
        "SET ae.task_id = $tid, ae.action_id = $aid, ae.step = $step, "
        "    ae.n_cells_changed = $ncc, ae.apparent_effect = $eff, ae.created_at = timestamp()",
        {"eid": effect_id, "tid": task_id, "aid": action_id, "step": step,
         "ncc": effect.get("n_cells_changed", 0), "eff": effect.get("apparent_effect", "unknown")}
    )
    
    # Upsert ActionFact
    fact_id = f"{task_id}_{action_id}"
    await db.execute_write(
        "MERGE (af:ActionFact {fact_id: $fid}) "
        "SET af.task_id = $tid, af.action_id = $aid, "
        "    af.observation_count = coalesce(af.observation_count, 0) + 1, "
        "    af.last_updated = timestamp()",
        {"fid": fact_id, "tid": task_id, "aid": action_id}
    )
    
    return {"status": "ok", "fact_id": fact_id, "effect_id": effect_id}


async def arc_get_entity_movement(params: dict, db, config: dict) -> dict:
    """Track entity positions relative to goal across steps.
    params: {task_id, step}
    """
    task_id = params["task_id"]
    step = params["step"]
    
    result = db.execute(
        "MATCH (ge:GridEntity {task_id: $tid})-[m:MOVED_BY]->(ae:ActionEffect {step: $step}) "
        "RETURN ge.entity_id, m.delta_row, m.delta_col",
        {"tid": task_id, "step": step})
    
    entities = []
    while result.has_next():
        row = result.get_next()
        entities.append({"id": row[0], "delta_row": row[1] or 0, "delta_col": row[2] or 0,
                         "moved_toward_goal": False, "distance_delta": 0})
    
    return {"entities": entities}

async def arc_get_goal_evidence(params: dict, db, config: dict) -> dict:
    """Query VictoryCondition nodes with supporting/contradicting evidence.
    params: {task_id}
    """
    task_id = params["task_id"]
    
    result = db.execute(
        "MATCH (vc:VictoryCondition {task_id: $tid}) "
        "OPTIONAL MATCH (vc)<-[s:INFERRED_FROM]-(h:Hypothesis) "
        "RETURN vc.condition_id, vc.condition_type, vc.confidence, "
        "       count(CASE WHEN h.status = 'active' THEN 1 END) as supports, "
        "       count(CASE WHEN h.status = 'demoted' THEN 1 END) as contradicts",
        {"tid": task_id})
    
    goals = []
    while result.has_next():
        row = result.get_next()
        goals.append({"id": row[0], "type": row[1], "confidence": row[2] or 0,
                       "supports": row[3] or 0, "contradicts": row[4] or 0})
    
    return {"goals": goals}


async def arc_classify_game_archetype(params: dict, db, config: dict) -> dict:
    """Match current game to known archetypes via Concept graph.
    params: {task_id, grid_features: {rows, cols, n_colors, symmetry}}
    """
    # Use vector search over Concept nodes to find matching archetypes
    features = params.get("grid_features", {})
    query_text = f"grid game {features.get('n_colors', 0)} colors {features.get('symmetry', 'none')} symmetry"
    
    try:
        from campy.brain.hippocampus.graph.embeddings import embed
        vec = embed(query_text)
        results = db.vector_search("Concept", "concept_embedding", vec, limit=3)
        concepts = [{"name": r.get("node", {}).get("text_raw", ""), 
                      "confidence": r.get("score", 0)} for r in results]
        archetype = concepts[0]["name"] if concepts else "unknown"
        return {"archetype": archetype, "confidence": concepts[0]["confidence"] if concepts else 0,
                "matching_concepts": concepts}
    except Exception:
        return {"archetype": "unknown", "confidence": 0, "matching_concepts": []}


async def arc_confirm_hypothesis(params: dict, db, config: dict) -> dict:
    """Write: evidence supports this hypothesis.
    params: {task_id, hypothesis_id, evidence: {description, weight}}
    """
    hyp_id = params["hypothesis_id"]
    weight = params.get("evidence", {}).get("weight", 1.0)
    
    await db.execute_write(
        "MATCH (h:Hypothesis {id: $hid}) "
        "SET h.evidence_count = coalesce(h.evidence_count, 0) + 1, "
        "    h.confidence = min(1.0, coalesce(h.confidence, 0.5) + $boost)",
        {"hid": hyp_id, "boost": min(0.1, weight * 0.05)}
    )
    
    result = db.execute("MATCH (h:Hypothesis {id: $hid}) RETURN h.confidence", {"hid": hyp_id})
    new_conf = result.get_next()[0] if result.has_next() else 0
    
    return {"status": "ok", "new_confidence": new_conf}


async def arc_contradict_hypothesis(params: dict, db, config: dict) -> dict:
    """Write: evidence contradicts this hypothesis.
    params: {task_id, hypothesis_id, evidence: {description, weight}}
    """
    hyp_id = params["hypothesis_id"]
    weight = params.get("evidence", {}).get("weight", 1.0)
    
    await db.execute_write(
        "MATCH (h:Hypothesis {id: $hid}) "
        "SET h.evidence_count = coalesce(h.evidence_count, 0) + 1, "
        "    h.confidence = max(0.0, coalesce(h.confidence, 0.5) - $penalty)",
        {"hid": hyp_id, "penalty": min(0.15, weight * 0.1)}
    )
    
    result = db.execute("MATCH (h:Hypothesis {id: $hid}) RETURN h.confidence, h.status", {"hid": hyp_id})
    if result.has_next():
        row = result.get_next()
        falsified = (row[0] or 0) < 0.1
        if falsified:
            await db.execute_write(
                "MATCH (h:Hypothesis {id: $hid}) SET h.status = 'demoted'", {"hid": hyp_id})
        return {"status": "ok", "new_confidence": row[0], "falsified": falsified}
    
    return {"status": "ok", "new_confidence": 0, "falsified": False}


async def arc_update_goal_confidence(params: dict, db, config: dict) -> dict:
    """Write: update goal confidence with grounding gate.
    params: {task_id, goal_id, new_confidence, has_meaningful_progress}
    """
    goal_id = params["goal_id"]
    new_conf = params["new_confidence"]
    has_progress = params.get("has_meaningful_progress", False)
    
    # Grounding gate: confidence can only increase with meaningful progress
    result = db.execute(
        "MATCH (vc:VictoryCondition {condition_id: $gid}) RETURN vc.confidence", {"gid": goal_id})
    
    if result.has_next():
        current = result.get_next()[0] or 0
        if new_conf > current and not has_progress:
            new_conf = current  # gate: hold steady
    
    await db.execute_write(
        "MATCH (vc:VictoryCondition {condition_id: $gid}) SET vc.confidence = $conf",
        {"gid": goal_id, "conf": new_conf})
    
    return {"status": "ok", "gated_confidence": new_conf}


async def arc_get_mechanic_priors(params: dict, db, config: dict) -> dict:
    """Cross-game transfer: matching mechanics from past games.
    params: {task_id, action_patterns: [], game_features: {}}
    """
    result = db.execute(
        "MATCH (m:ArcMechanic)-[:ARC_MECHANIC_HAS_ACTION_PATTERN]->(ap:ArcActionPattern) "
        "WHERE m.confidence > 0.3 "
        "RETURN m.mechanic_id, m.name, m.confidence, ap.signature, ap.action_set "
        "ORDER BY m.confidence DESC LIMIT 5")
    
    mechanics = []
    while result.has_next():
        row = result.get_next()
        mechanics.append({"id": row[0], "name": row[1], "confidence": row[2],
                          "action_signature": row[3], "action_set": row[4]})
    
    return {"mechanics": mechanics}

async def arc_check_action_gate(params: dict, db, config: dict) -> dict:
    """Go/No-Go decision for an ARC action.
    params: {task_id, action_id, available_actions: []}
    """
    task_id = params["task_id"]
    action_id = params["action_id"]
    available = params.get("available_actions", [])
    
    # Query ActionFact for falsification history
    result = db.execute(
        "MATCH (af:ActionFact {task_id: $tid, action_id: $aid}) "
        "RETURN af.confidence, af.value_status, af.observation_count",
        {"tid": task_id, "aid": action_id})
    
    falsification_count = 0
    if result.has_next():
        row = result.get_next()
        if (row[1] or "") in ("harmful", "no_effect"):
            falsification_count = row[2] or 0
    
    # Check untested
    untested_result = await arc_get_untested_actions(
        {"task_id": task_id, "available_actions": available}, db, config)
    untested_available = len(untested_result.get("untested", [])) > 0
    
    # Go/No-Go
    go = True
    reason = "approved"
    
    if falsification_count >= 3 and untested_available:
        go = False
        reason = f"{action_id} falsified {falsification_count} times, untested alternatives exist"
    
    # Also call B277 general gate for Procedure-level evidence
    try:
        from campy.brain.basal_ganglia.action_selector import check_action_gate as general_gate
        general_result = await general_gate(db, action_id, task_id, domain="arc")
        if general_result.get("decision") == "no_go" and go:
            go = False
            reason = general_result.get("reason", "basal_ganglia_no_go")
    except Exception:
        pass  # B277 gate is supplementary
    
    return {"go": go, "reason": reason, "falsification_count": falsification_count,
            "reward_prediction_error": 0, "untested_available": untested_available}


async def arc_record_reward_prediction_error(params: dict, db, config: dict) -> dict:
    """Write: predicted vs actual reward delta.
    params: {task_id, action_id, step, predicted_reward, actual_reward}
    """
    task_id = params["task_id"]
    action_id = params["action_id"]
    predicted = params.get("predicted_reward", 0)
    actual = params.get("actual_reward", 0)
    error = actual - predicted
    
    # Record on ActionFact
    fact_id = f"{task_id}_{action_id}"
    await db.execute_write(
        "MATCH (af:ActionFact {fact_id: $fid}) "
        "SET af.predicted_valence = $pred, af.actual_valence = $act, "
        "    af.prediction_error = $err",
        {"fid": fact_id, "pred": predicted, "act": actual, "err": error}
    )
    
    # Also record via B277 general RPE tracker
    try:
        from campy.brain.basal_ganglia.reward_predictor import record_reward_prediction_error
        await record_reward_prediction_error(db, fact_id, predicted, actual)
    except Exception:
        pass
    
    return {"status": "ok", "prediction_error": error,
            "direction": "positive" if error > 0.1 else "negative" if error < -0.1 else "neutral"}
```

### Notes

- ARC tools use the same `(params, db, config)` signature as all other tool handlers
- ARC tools do NOT use `_with_phase()` wrapping — they're stateless query tools
- B277's basal ganglia modules are called as supplementary signals, not primary
- The primary action gating for ARC is based on `ActionFact` contradiction counts (task-scoped), not Procedure vector search (global)
- All tools are designed to degrade gracefully — empty results on missing data, never crash

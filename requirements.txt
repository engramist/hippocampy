# Embeddings (384-dim — matches FLOAT[384] schema; changing requires full re-embedding)
# B355: fastembed/ONNX Runtime replaced sentence-transformers/PyTorch as the
# local embedding backend -- same model (sentence-transformers/all-MiniLM-
# L6-v2, via its ONNX conversion, qdrant/all-MiniLM-L6-v2-onnx), verified
# near-perfect output parity (cosine similarity 1.000000 across 200 real
# test sentences). Motivated by B342/B353: PyTorch/sentence-transformers
# was the dominant contributor to brain_daemon.py's ~1.2-2GB fresh-start
# memory footprint; fastembed runs the identical model at ~60-80MB. This
# also resolves the transformers-CVE pinning conflict the previous
# sentence-transformers==3.3.1 pin was stuck with (fastembed does not
# depend on the `transformers` package at all). See backlog/B353.md and
# backlog/B355.md for the full investigation and verification.
fastembed==0.8.0

# AST code folding (campy/brain/thalamus/compression/ast_mapper.py) — pulls
# in tree-sitter transitively. Already declared in pyproject.toml's main
# `dependencies`, but that alone doesn't help a plain
# `pip install -r requirements.txt` (this file's own stated purpose, see
# the boto3 comment above): without it, ASTCodeCompressor.compress()
# silently falls back to its documented "tree-sitter unavailable" no-op
# path (by design — see _get_python_parser()'s docstring), which fails
# tests/test_compression_ast.py and the ast_code case in
# tests/test_compression_regression.py deterministically. CI's own
# tests.yml only avoided this because it separately runs `pip install -e .`
# (which reads pyproject.toml) right after this file; a `-r requirements.txt`
# -only install — the natural first thing anyone evaluating this repo would
# run — did not.
tree-sitter-languages>=1.10.0

# Web server (Memory Control Panel — M7)
# fastapi bumped 0.115.6 -> 0.133.0 (security gate: starlette HIGH CVEs
# PYSEC-2026-{161,248,249,1942,1941,2281,2280}, fixes require starlette
# >=1.0.1..>=1.3.1). starlette is a transitive dep, not listed directly
# below, but fastapi==0.115.6 pins it to <0.42.0 in its own metadata, so a
# direct starlette override can't coexist with fastapi==0.115.6 (pip would
# hit a hard resolution conflict). 0.133.0 is the first fastapi release
# that drops the starlette upper bound entirely (0.132.0 still caps it at
# <1.0.0) — smallest fastapi bump that unblocks the fix. No on_event/other
# deprecated APIs used in web/server.py; full pytest + live-server smoke
# test passed post-bump (see commit message).
fastapi==0.133.0
# Direct pin (transitive via fastapi) to satisfy the highest starlette CVE
# fix floor (>=1.3.1) explicitly, per "pin all versions" policy above.
starlette==1.6.0
uvicorn==0.32.1
